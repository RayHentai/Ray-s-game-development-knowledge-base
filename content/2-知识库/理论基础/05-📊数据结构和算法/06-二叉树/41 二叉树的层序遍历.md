## 层序遍历（迭代）

**思路：**

**时间复杂度：** O()
**空间复杂度：** O()

```csharp
public class Solution {
    public IList<IList<int>> LevelOrder(TreeNode root) {
        Queue<TreeNode> queue = new();
        List<IList<int>> result = new();
        if (root == null)
            return result;
        queue.Enqueue(root);
        while (queue.Count > 0){
            TreeNode node;
            int index = queue.Count;
            List<int> list = new();
            for (int i = 0; i < index; i++){
                node = queue.Dequeue();
                list.Add(node.val);
                if (node.left != null)
                    queue.Enqueue(node.left);
                if (node.right != null)
                    queue.Enqueue(node.right);
            }
            result.Add(list);
        }
        return result;
    }
}
```

## 层序遍历（递归）

**核心思路：**

**时间复杂度：** O(n)
**空间复杂度：** O(n)

```csharp
public class Solution {
    public IList<IList<int>> LevelOrder(TreeNode root) {
        List<IList<int>> result = new();
        traversal(root, 0, result);
        return result;
        void traversal(TreeNode root, int depth, List<IList<int>> result){
            if (root == null)
                return;
            if (result.Count == depth)
                result.Add(new List<int>());
            result[depth].Add(root.val);
            traversal(root.left, depth + 1, result);
            traversal(root.right, depth + 1, result);
        }
    }
}
```

## 层倒序遍历（递归）

## 二叉树的右视图（递归）

**核心思路：递归函数中先递归右子树再递归左子树，这样每层第一个记录的就是右边的元素**

**时间复杂度：** O(n)
**空间复杂度：** O(n)

```csharp
public class Solution {
    public IList<int> RightSideView(TreeNode root) {
        var result = new List<int>();
        traversal(root);
        return result;
        void traversal(TreeNode root, int depth = 0){
            if (root == null)
                return;
            if (result.Count == depth)
                result.Add(root.val);
            traversal(root.right, depth + 1);
            traversal(root.left, depth + 1);
        }
    }
}
```

## 二叉树的右视图（迭代）

**时间复杂度：** O(n)

```csharp
public class Solution {
    public IList<int> RightSideView(TreeNode root) {

        List<int> result = new();
        if (root == null)
            return result;

        while (queue.Count > 0){
            TreeNode node;
            int index = queue.Count;
            for (int i = 0; i < index; i++){
                node = queue.Dequeue();
                if (i == 0)
                    result.Add(node.val);

                    queue.Enqueue(node.right);
                if (node.left != null)
                    queue.Enqueue(node.left);
            }
        }
        return result;
    }
}
```

## 二叉树的层平均值（递归）

**核心思路：声明两个List存储每层元素数量和总和，递归函数执行完毕后算平均数**

**时间复杂度：** O(n)
**空间复杂度：** O(n)

```csharp
public class Solution {
    public IList<double> AverageOfLevels(TreeNode root) {
        var result = new List<double>();
        var count = new List<int>();
        var sum = new List<long>();
        traversal(root);
        for (int i = 0; i < sum.Count; i++)
            result.Add((double)sum[i] / count[i]);
        return result;
        void traversal(TreeNode root, int depth = 0){
            if (root == null)
                return;
            if (sum.Count == depth){
                count.Add(1);
                sum.Add(root.val);
            }
            else{
                ++count[depth];
                sum[depth] += root.val;
            }
            traversal(root.left, depth + 1);
            traversal(root.right, depth + 1);
        }
    }
}
```

## 二叉树的层平均值（迭代）

**核心思路：**

**时间复杂度：** O(n)
**空间复杂度：** O(n)

```csharp
public class Solution {
    public IList<double> AverageOfLevels(TreeNode root) {
        var result = new List<Double>();
        if (root == null)
            return result;
        var queue = new Queue<TreeNode>();
        queue.Enqueue(root);
        while (queue.Count > 0){
            int index = queue.Count;
            TreeNode node;
            long sum = 0;
            for (int i = 0; i < index; i++){
                node = queue.Dequeue();
                sum += node.val;
                if (node.left != null)
                    queue.Enqueue(node.left);
                if (node.right != null)
                    queue.Enqueue(node.right);
            }
            result.Add((double)sum / index);
        }
        return result;
    }
}
```

## 在每个树行中找最大值（递归）

**核心思路：**

**时间复杂度：** O(n)
**空间复杂度：** O(n)

```csharp
public class Solution {
    public IList<int> LargestValues(TreeNode root) {
        var result = new List<int>();
        traversal(root);
        return result;
        void traversal(TreeNode root, int depth = 0){
            if (root == null)
                return;
            if (result.Count == depth)
                result.Add(root.val);
            else if (result[depth] < root.val){
                result[depth] = root.val;
            }
            traversal(root.left, depth + 1);
            traversal(root.right, depth + 1);
        }
    }
}
```

## 在每个树行中找最大值（迭代）

**核心思路：**

**时间复杂度：** O(n)
**空间复杂度：** O(n)

```csharp
public class Solution {
    public IList<int> LargestValues(TreeNode root) {
        var result = new List<int>();
        if (root == null)
            return result;
        var queue = new Queue<TreeNode>();
        queue.Enqueue(root);
        while (queue.Count > 0){
            int index = queue.Count;
            int max = int.MinValue;
            TreeNode node;
            for (int i = 0; i < index; i++){
                node = queue.Dequeue();
                if (node.val > max)
                    max = node.val;
                if (node.left != null)
                    queue.Enqueue(node.left);
                if (node.right != null)
                    queue.Enqueue(node.right);
            }
            result.Add(max);
        }
        return result;
    }
}
```

## 填充每个节点的下一个右侧节点指针（双指针+递归）

**核心思路：**

**时间复杂度：** O(n)
**空间复杂度：** O(n)

```csharp
public class Solution {
    public Node Connect(Node root) {
        Node levelStart = root;
        Node cur = root;
        traversal(levelStart);
        return root;
        void traversal(Node levelStart){
            if (levelStart == null || levelStart.left == null)
                return;
            cur = levelStart;
            while (cur != null){
                cur.left.next = cur.right;
                if (cur.next != null)
                    cur.right.next = cur.next.left;
                cur = cur.next;
            }
            traversal(levelStart.left);
        }
    }
}
```

## 填充每个节点的下一个右侧节点指针（双指针+迭代）

**核心思路：用两个指针维护两层循环**

**时间复杂度：** O(n)
**空间复杂度：** O(1)

```csharp
public class Solution {
    public Node Connect(Node root) {
        Node levelStart = root;
        Node cur = root;
        while (levelStart != null && levelStart.left != null){//后面的条件排除只有一层的情况
            while (cur != null){
                cur.left.next = cur.right;
                if (cur.next != null)
                    cur.right.next = cur.next.left;
                cur = cur.next;
            }
            levelStart = levelStart.left;
            cur = levelStart;
        }
        return root;
    }
}
```

## 填充每个节点的下一个右侧节点指针 II（迭代）

**核心思路：**

**时间复杂度：** O(n)
**空间复杂度：** O(n)

```csharp
public class Solution {
    public Node Connect(Node root) {
        if (root == null)
            return root;
        var queue = new Queue<Node>();
        queue.Enqueue(root);
        while(queue.Count > 0){
            int index = queue.Count;
            Node node;
            for (int i = 0; i < index; i++){
                node = queue.Dequeue();
                if (i != index - 1)
                    node.next = queue.Peek();
                if (node.left != null)
                    queue.Enqueue(node.left);
                if (node.right != null)
                    queue.Enqueue(node.right);
            }
        }
        return root;
    }
}
```
